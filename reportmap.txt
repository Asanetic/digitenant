report Collections "Rent Collections"
  nest rent
  by month|day|year
  date payment_date
  entity rental_rent_collections
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter method = payment_method

  # payment_status values in your data: COMPLETED
  payments = count rent_payments where payment_status=COMPLETED "Payments"
  collected = sum rent_payments.amount where payment_status=COMPLETED "Rent Collected" money
  avg_payment = collected / payments "Average Payment" money
  chg = change(collected) "Collected vs Previous"

  forecast collected next 3
  forecast payments next 3

  headline {label}: {payments} payments, {collected} collected[, {updown} {abs(chg)}% on {prev_label}].
  remark payments == 0 | watch | No payments | No rent payments were received in {label}.
  remark chg !== null && chg <= -20 | watch | Collections dropped | {label}: {collected} collected, down {abs(chg)}% on {prev_label}.
  remark true | neutral | Collections | {payments} payments totalling {collected}, averaging {avg_payment}.


report Billing "Billed vs Collected"
  nest rent
  by month|year
  date invoice_date
  entity rental_rent_billing
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  filter status = invoice_status
  filter property = property_id

  # outstanding is worked out as billed minus paid, so it does not depend on balance_amount being filled
  invoices = count rent_invoices "Invoices Issued"
  billed = sum rent_invoices.total_amount "Billed" money
  collected = sum rent_invoices.paid_amount "Collected" money
  outstanding = billed - collected "Outstanding" money
  rate = round(collected / billed * 100) "Collected %"
  chg = change(billed) "Billed vs Previous"

  forecast billed next 3

  headline {label}: {billed} billed, {rate}% collected, {outstanding} outstanding.
  remark billed == 0 | neutral | Nothing billed | No invoices were issued in {label}.
  remark rate !== null && rate < 70 | bad | Weak collection | Only {rate}% of the {billed} billed in {label} was collected, and {outstanding} is outstanding.
  remark rate !== null && rate < 90 | watch | Gaps | {rate}% of the {billed} billed in {label} was collected, with {outstanding} still outstanding.
  remark true | good | On track | {rate}% of the {billed} billed in {label} was collected.


report Tenants "Arrears by Tenant"
  nest arrears
  by dimension
  date due_date
  entity rental_arrears_tenants
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  group tenant_id
  rows "Tenant"
  labels tenants_list.tenant_id > tenant_name
  sort owed desc

  # Set the date range to end today: it covers invoices that fell due in that range.
  # Owed = billed minus paid, so it works without balance_amount or invoice_status.
  invoices = count rent_invoices "Invoices Due"
  unpaid_inv = count rent_invoices where paid_amount=0 "Nothing Paid"
  billed = sum rent_invoices.total_amount "Billed" money
  paid = sum rent_invoices.paid_amount "Paid" money
  owed = billed - paid "Balance Owed" money
  share = share(owed) "Share of Arrears"

  headline {label} owes the most: {owed} across {invoices} invoices, {unpaid_inv} with nothing paid.
  remark unpaid_inv > 0 | bad | Not paying | {label} has {unpaid_inv} invoices with nothing paid and owes {owed} in total.
  remark owed > 0 | watch | Balance owed | {label} owes {owed} of the {billed} billed.
  remark true | good | Up to date | {label} has paid everything due in this period.

  maxrange 1200

  group property_id
  rows "Property"
  labels properties_list.property_id > property_name

  # payments reach the property through their invoice (rent_invoices.property_id)
  payments = count rent_payments where payment_status=COMPLETED via rent_invoices on invoice_id=invoice_id by=payment_date "Payments"

  billed = sum rent_invoices.total_amount by=invoice_date "Billed" money
  rate = round(collected / billed * 100) "Collected %"
  share = share(collected) "Share of Income"

  headline {label} earns the most: {collected} from {payments} payments, {share}% of all rent collected.
  remark collected == 0 | bad | No income | {label} received no rent in this period.
  remark rate !== null && rate < 70 | watch | Low collection | {label}: only {rate}% of the {billed} billed was collected.
  remark true | good | Performing | {label}: {collected} collected from {payments} payments, {share}% of total income.

report Statement "Landlord Payouts"
  nest landlords
  by dimension
  date period_end
  entity rental_landlords_statement
  role view_reports
  currency KES
  cache 60
  maxrange 1200

  group landlord_id
  rows "Landlord"
  labels properties_landlords.landlord_id > landlord_name
  sort net desc

  # reads properties_payouts, so it stays empty until payouts are generated
  # assumes payout_status=PAID once the landlord has been remitted
  statements = count properties_payouts "Statements"
  collected = sum properties_payouts.collected_amount "Rent Collected" money
  expenses = sum properties_payouts.expenses_amount "Expenses" money
  commission = sum properties_payouts.commission_amount "Commission" money
  net = sum properties_payouts.net_amount "Net Payable" money
  paid_out = sum properties_payouts.net_amount where payout_status=PAID "Paid Out" money
  owed = net - paid_out "Still Owed" money
  share = share(net) "Share of Payouts"

  headline {label} has the largest payout: {net} net from {collected} collected, {owed} still owed.
  remark owed > 0 | watch | Payout due | {label}: {owed} still to be paid out of {net} net.
  remark true | good | Settled | {label}: {net} net paid out after {commission} commission and {expenses} expenses.
